PRODUCT OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
DEHIX Live Room is a real-time AI-powered hiring session 
where a business goes from a raw project idea to a fully 
assembled, contracted Web3 team in a single sitting — 
under 60 minutes.

The core flow:
1. Business enters a raw project description in plain text
2. AI scopes it into structured roles, tickets, milestones
3. AI matches verified available talent in real time
4. Talent joins the Live Room (video + chat)
5. Business reviews verified credentials, asks questions
6. One click assembles the squad + generates NDA + opens 
   project board

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

-- Users (both talent and businesses)
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT UNIQUE NOT NULL,
  role TEXT CHECK (role IN ('talent','business')) NOT NULL,
  name TEXT NOT NULL,
  avatar_url TEXT,
  wallet_address TEXT UNIQUE,
  is_online BOOLEAN DEFAULT false,
  last_seen TIMESTAMPTZ DEFAULT NOW(),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- SBT Credentials (Soulbound Token records)
CREATE TABLE sbt_credentials (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id),
  skill_domain TEXT NOT NULL,
  level INTEGER CHECK (level IN (1,2)) NOT NULL,
  reputation_score INTEGER DEFAULT 0 CHECK 
    (reputation_score BETWEEN 0 AND 1000),
  status TEXT CHECK (status IN 
    ('verified','disputed','revoked')) DEFAULT 'verified',
  github_score INTEGER DEFAULT 0,
  interview_score INTEGER DEFAULT 0,
  projects_completed INTEGER DEFAULT 0,
  on_chain_tx TEXT,
  issued_at TIMESTAMPTZ DEFAULT NOW(),
  embedding vector(1536)  -- for pgvector matching
);

-- Live Rooms
CREATE TABLE live_rooms (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_code TEXT UNIQUE NOT NULL,
  business_id UUID REFERENCES users(id),
  title TEXT NOT NULL,
  raw_description TEXT NOT NULL,
  ai_scoped_brief JSONB,
  status TEXT CHECK (status IN (
    'scoping','matching','open',
    'assembling','contracted','closed'
  )) DEFAULT 'scoping',
  daily_room_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  contracted_at TIMESTAMPTZ
);

-- Room Roles (AI-generated required roles)
CREATE TABLE room_roles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id UUID REFERENCES live_rooms(id),
  role_title TEXT NOT NULL,
  skill_domain TEXT NOT NULL,
  required_level INTEGER DEFAULT 1,
  min_reputation INTEGER DEFAULT 0,
  filled_by UUID REFERENCES users(id),
  status TEXT CHECK (status IN (
    'open','invited','accepted','filled'
  )) DEFAULT 'open'
);

-- Room Participants (talent in the room)
CREATE TABLE room_participants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id UUID REFERENCES live_rooms(id),
  user_id UUID REFERENCES users(id),
  role_id UUID REFERENCES room_roles(id),
  joined_at TIMESTAMPTZ DEFAULT NOW(),
  status TEXT CHECK (status IN (
    'invited','joined','accepted','declined'
  )) DEFAULT 'invited'
);

-- Tickets (AI-generated project tasks)
CREATE TABLE tickets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id UUID REFERENCES live_rooms(id),
  title TEXT NOT NULL,
  description TEXT,
  assigned_role UUID REFERENCES room_roles(id),
  milestone_number INTEGER DEFAULT 1,
  estimated_hours INTEGER,
  status TEXT CHECK (status IN (
    'backlog','todo','in_progress','done'
  )) DEFAULT 'backlog',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Milestones + Escrow
CREATE TABLE milestones (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id UUID REFERENCES live_rooms(id),
  title TEXT NOT NULL,
  description TEXT,
  amount_usd NUMERIC(10,2),
  due_date DATE,
  status TEXT CHECK (status IN (
    'pending','in_progress','submitted',
    'approved','released'
  )) DEFAULT 'pending'
);

-- Chat Messages
CREATE TABLE chat_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id UUID REFERENCES live_rooms(id),
  user_id UUID REFERENCES users(id),
  message TEXT NOT NULL,
  is_ai BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- NDA Records
CREATE TABLE ndas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id UUID REFERENCES live_rooms(id),
  content TEXT NOT NULL,
  signed_by UUID[] DEFAULT '{}',
  status TEXT CHECK (status IN (
    'draft','pending_signatures','signed'
  )) DEFAULT 'draft',
  created_at TIMESTAMPTZ DEFAULT NOW()
);
